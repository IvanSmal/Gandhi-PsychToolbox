classdef SoundGenerator < matlab.apps.AppBase

    % Properties that correspond to app components
    properties (Access = public)
        UIFigure                 matlab.ui.Figure
        Button                   matlab.ui.control.StateButton
        WhiteNoiseCheckBox       matlab.ui.control.CheckBox
        SineCheckBox             matlab.ui.control.CheckBox
        DurationsEditField       matlab.ui.control.NumericEditField
        DurationsEditFieldLabel  matlab.ui.control.Label
        AmplitudeEditField       matlab.ui.control.NumericEditField
        AmplitudeEditFieldLabel  matlab.ui.control.Label
        FrequencyEditField       matlab.ui.control.NumericEditField
        FrequencyEditFieldLabel  matlab.ui.control.Label
    end


    properties (Access = private)
        PlayThis; % sound to be played
        soundGenerator=audioDeviceWriter('SampleRate',44100);
        commport;
        duration;
        commid_history=0;
    end

    methods (Access = private)

        function app = GenerateSound_gui(app)
            issine=app.SineCheckBox.Value;
            if issine
            app.PlayThis = dsp.SineWave(app.AmplitudeEditField.Value,...
                app.FrequencyEditField.Value,...
                'SampleRate', 44100,...
                'SamplesPerFrame', 128);
            else
            app.PlayThis = dsp.ColoredNoise('Color','white',...
                'SamplesPerFrame', 128);
            end
            app.duration=app.DurationsEditField.Value;
        end

        function PlaySound(app)
            tic;
            while toc<app.duration
                app.soundGenerator(app.PlayThis());
            end
            app.soundGenerator.release
            app.Button.Value=0;
        end
        
        function getCommands(app,~,~)
            command=readline(app.commport);
            try
                eval(command)
            end
        end
        
        function GenerateSound_udp(app, commid, type, dur, amp, hz)            
            if app.commid_history ~=commid
                app.commid_history = commid;
                if type==1
                app.PlayThis = dsp.SineWave(amp,...
                    hz,...
                    'SampleRate', 44100,...
                    'SamplesPerFrame', 128);
                elseif type==2
                app.PlayThis = dsp.ColoredNoise('Color','white',...
                    'SamplesPerFrame', 128);
                elseif type==3
                    app.PlayThis = dsp.AudioFileReader('/home/gandhi/Documents/MATLAB/Gandhi-PsychToolbox/Main code/assets/start.mp3');
                elseif type==4
                    app.PlayThis = dsp.AudioFileReader('/home/gandhi/Documents/MATLAB/Gandhi-PsychToolbox/Main code/assets/reward.mp3');
                end  
                app.duration=dur;
                app.PlaySound()
            end
        end
    end


    % Callbacks that handle component events
    methods (Access = private)

        % Code that executes after component creation
        function startupFcn(app)

            % system('echo 4096 | /usr/bin/tee /proc/asound/card0/pcm0p/sub0/prealloc')
            app.commport = udpport("LocalPort",2025);
            configureCallback(app.commport,"terminator",@app.getCommands);
            GenerateSound_udp(app, randi(20), 1, 30, 1, 20)  ;
            clc;
            disp('-----------------Sound Generator-----------------')

        end

        % Value changed function: FrequencyEditField
        function FrequencyEditFieldValueChanged(app, event)
            app.GenerateSound_gui;
        end

        % Value changed function: Button
        function ButtonValueChanged(app, event)
            app.GenerateSound_gui;
            app.PlaySound
        end

        % Value changed function: AmplitudeEditField
        function AmplitudeEditFieldValueChanged(app, event)
            value = app.AmplitudeEditField.Value;
            app.GenerateSound_gui;
        end

        % Value changed function: DurationsEditField
        function DurationsEditFieldValueChanged(app, event)
            value = app.DurationsEditField.Value;
            app.GenerateSound_gui;
        end

        % Value changed function: WhiteNoiseCheckBox
        function WhiteNoiseCheckBoxValueChanged(app, event)
            app.SineCheckBox.Value=0;
            app.FrequencyEditField.Enable=0;
            app.GenerateSound_gui;
        end

        % Value changed function: SineCheckBox
        function SineCheckBoxValueChanged(app, event)
            app.WhiteNoiseCheckBox.Value=0;
            app.FrequencyEditField.Enable=1;
            app.GenerateSound_gui;
        end

        % Close request function: UIFigure
        function UIFigureCloseRequest(app, event)
            delete(app)
            
        end
    end

    % Component initialization
    methods (Access = private)

        % Create UIFigure and components
        function createComponents(app)

            % Create UIFigure and hide until all components are created
            app.UIFigure = uifigure('Visible', 'off');
            app.UIFigure.Position = [100 100 245 285];
            app.UIFigure.Name = 'MATLAB App';
            app.UIFigure.CloseRequestFcn = createCallbackFcn(app, @UIFigureCloseRequest, true);

            % Create FrequencyEditFieldLabel
            app.FrequencyEditFieldLabel = uilabel(app.UIFigure);
            app.FrequencyEditFieldLabel.HorizontalAlignment = 'right';
            app.FrequencyEditFieldLabel.Position = [28 169 62 22];
            app.FrequencyEditFieldLabel.Text = 'Frequency';

            % Create FrequencyEditField
            app.FrequencyEditField = uieditfield(app.UIFigure, 'numeric');
            app.FrequencyEditField.ValueChangedFcn = createCallbackFcn(app, @FrequencyEditFieldValueChanged, true);
            app.FrequencyEditField.Position = [105 169 100 22];
            app.FrequencyEditField.Value = 200;

            % Create AmplitudeEditFieldLabel
            app.AmplitudeEditFieldLabel = uilabel(app.UIFigure);
            app.AmplitudeEditFieldLabel.HorizontalAlignment = 'right';
            app.AmplitudeEditFieldLabel.Position = [32 128 58 22];
            app.AmplitudeEditFieldLabel.Text = 'Amplitude';

            % Create AmplitudeEditField
            app.AmplitudeEditField = uieditfield(app.UIFigure, 'numeric');
            app.AmplitudeEditField.ValueChangedFcn = createCallbackFcn(app, @AmplitudeEditFieldValueChanged, true);
            app.AmplitudeEditField.Position = [105 128 100 22];
            app.AmplitudeEditField.Value = 1;

            % Create DurationsEditFieldLabel
            app.DurationsEditFieldLabel = uilabel(app.UIFigure);
            app.DurationsEditFieldLabel.HorizontalAlignment = 'right';
            app.DurationsEditFieldLabel.Position = [22 87 68 22];
            app.DurationsEditFieldLabel.Text = 'Duration (s)';

            % Create DurationsEditField
            app.DurationsEditField = uieditfield(app.UIFigure, 'numeric');
            app.DurationsEditField.ValueChangedFcn = createCallbackFcn(app, @DurationsEditFieldValueChanged, true);
            app.DurationsEditField.Position = [105 87 100 22];
            app.DurationsEditField.Value = 2;

            % Create SineCheckBox
            app.SineCheckBox = uicheckbox(app.UIFigure);
            app.SineCheckBox.ValueChangedFcn = createCallbackFcn(app, @SineCheckBoxValueChanged, true);
            app.SineCheckBox.Text = 'Sine';
            app.SineCheckBox.Position = [44 219 46 22];
            app.SineCheckBox.Value = true;

            % Create WhiteNoiseCheckBox
            app.WhiteNoiseCheckBox = uicheckbox(app.UIFigure);
            app.WhiteNoiseCheckBox.ValueChangedFcn = createCallbackFcn(app, @WhiteNoiseCheckBoxValueChanged, true);
            app.WhiteNoiseCheckBox.Text = 'White Noise';
            app.WhiteNoiseCheckBox.Position = [117 219 87 22];

            % Create Button
            app.Button = uibutton(app.UIFigure, 'state');
            app.Button.ValueChangedFcn = createCallbackFcn(app, @ButtonValueChanged, true);
            app.Button.Text = '🎺';
            app.Button.FontSize = 36;
            app.Button.Position = [90 18 69 54];

            % Show the figure after all components are created
            app.UIFigure.Visible = 'on';
        end
    end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = SoundGenerator

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
    end
end
